{{- /* Images as Obsidian pastes them: ![caption](img/file.png), looked up next to the post or in the blog's img folder. A missing file fails the build. Width and height are written out so the page does not jump while loading. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := .Destination -}} {{- $size := "" -}} {{- if and (not $u.Scheme) (not (hasPrefix $u.Path "/")) -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (.PageInner.CurrentSection.Resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $size = printf ` width="%d" height="%d"` .Width .Height -}} {{- end -}} {{- else -}} {{- errorf "%s: image %q not found" .PageInner.File.Path .Destination -}} {{- end -}} {{- end -}} {{- $img := printf `` $src (.PlainText | htmlEscape) $size | safeHTML -}} {{- if .IsBlock -}} {{- else -}} {{ $img }} {{- end -}}
