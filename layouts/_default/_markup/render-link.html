{{- /* Links as Obsidian writes them: [text](Other post.md) resolves to that post, and a missing post fails the build. Links that leave the site are marked and open in a new tab. */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := .Destination -}} {{- $out := and (hasPrefix $u.Scheme "http") (ne $u.Host (urls.Parse site.BaseURL).Host) -}} {{- if and (not $u.Scheme) (strings.HasSuffix $u.Path ".md") -}} {{- with .PageInner.GetPage $u.Path -}} {{- $href = .RelPermalink -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href (anchorize .) }}{{ end -}} {{- else -}} {{- errorf "%s: link to %q, but that post does not exist (is it still a draft?)" .PageInner.File.Path .Destination -}} {{- end -}} {{- end -}} {{ .Text }} {{- /**/ -}}
